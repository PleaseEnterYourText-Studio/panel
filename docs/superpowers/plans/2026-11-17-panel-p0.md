# PEYT Panel P0 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Cloudflare Workers 上交付一个可日常使用的面板 P0：能登录、能录入服务、能自动探测、能看到每个服务的对象页与真实状态。

**Architecture:** 单 Worker（Hono）+ D1 作为唯一真相源 + Cron Trigger 每分钟分片探测。数据源抽象成 `Source` 接口，命令通道抽象成 `CommandChannel`（P0 为禁用实现）。前端为 React SPA，由 Workers Assets 托管，所有页面围绕「对象」组织。

**Tech Stack:** TypeScript 5.9 · Hono 4 · Wrangler 4 · D1 · React 19 · Vite 8 · Vitest 5 · `@cloudflare/vitest-pool-workers` · jose 6

## Global Constraints

- **每次提交后必须立刻 `git push` 到 `origin`（https://github.com/PleaseEnterYourText-Studio/panel），哪怕只是一个小的提交。禁止只在本地攒着。**
- 运行时：**全部逻辑必须运行在 Cloudflare Workers**，不得引入常驻服务器、容器或本地守护进程。
- 兼容日期：`compatibility_date = "2026-11-01"`。
- 包版本（**as-built，以仓库实际安装为准**）：`wrangler@4.163.0`、`hono@4.13.19`、`react@19.3.1`、`react-dom@19.3.1`、`vite@8.3.8`、`vitest@4.1.12`、`typescript@5.9.3`、`jose@6.2.17`、`@cloudflare/vitest-pool-workers@0.23.0`、`@vitejs/plugin-react@6.3.0`、`@types/react@19.3.0`、`@types/react-dom@19.3.0`。**不要引入图表库、不要引入 ORM。**
  - `vitest` 为 4.1.12 而非 5.1.0：`@cloudflare/vitest-pool-workers@0.23.0` 的 peer 是 `^4.1.0`，5.x 装不上。
  - `@types/react(-dom)` 为 19.3.0：19.3.1 在 registry 不存在。
  - 该包已弃用 `defineWorkersConfig`，改用 `cloudflareTest` 插件 API；并已在 `vitest.config.ts` 覆盖 `compatibilityDate: '2026-08-22'`（其内嵌 workerd 上限），生产仍用 `wrangler.jsonc` 的 `2026-11-01`。差异已写进 README，迁移到 `@cloudflare/vitest-plugin` 后即可删除该覆盖。
- 探测默认间隔 `180` 秒；单次 Cron 调用最多探测 **40** 个 target；出站并发上限 **6**。
- 角色仅三种：`owner` / `ops` / `viewer`；未在 `users` 表登记的 email 一律拒绝。
- 所有非 GET 请求必须写审计（`events` 表 `kind='action.*'`）。
- 阶段三动作必须显示 `P3` 徽章且禁用，不允许点后才报错。
- 视觉取值固定：画布 `#0d1117`、面板 `#161b22`、边框 `#30363d`、正文 `#c9d1d9`、弱化 `#8b949e`、链接 `#58a6ff`、正常 `#3fb950`、疑似 `#d29922`、故障 `#f85149`、主按钮 `#238636`。
- 状态不得只用颜色表达，必须有文字/形状冗余。
- TypeScript `strict: true`，禁止 `any`（`unknown` + 收窄代替）。

`src/db/queries.ts`（关键函数，全部 prepared statement，禁止字符串拼 SQL）：

```ts
export interface AssetRow { id: string; kind: string; name: string; slug: string; grp: string | null; owner_email: string | null; tags: string | null; repo: string | null; notes: string | null; enabled: number; created_at: number }
export interface TargetRow { id: string; asset_id: string; source: string; form: string; config: string; interval_sec: number; timeout_ms: number; enabled: number; next_run_at: number }
export interface CheckRow { id?: number; target_id: string; at: number; ok: number; latency_ms: number | null; error_kind: string | null; detail: string | null }

export async function insertAsset(db: D1Database, a: { id: string; kind: string; name: string; slug: string; grp: string | null; ownerEmail: string | null; repo: string | null; tags: string | null; notes: string | null; enabled: number; createdAt: number }) {
  await db.prepare(
    `INSERT INTO assets (id, kind, name, slug, grp, owner_email, tags, repo, notes, enabled, created_at)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`
  ).bind(a.id, a.kind, a.name, a.slug, a.grp, a.ownerEmail, a.tags, a.repo, a.notes, a.enabled, a.createdAt).run();
}

export async function getAssetBySlug(db: D1Database, slug: string) {
  return db.prepare(`SELECT * FROM assets WHERE slug = ?`).bind(slug).first<AssetRow>();
}

export async function listDueTargets(db: D1Database, now: number, limit: number) {
  return db.prepare(
    `SELECT * FROM targets WHERE enabled = 1 AND next_run_at <= ? ORDER BY next_run_at ASC LIMIT ?`
  ).bind(now, limit).all<TargetRow>();
}

export async function listDueTargetsBatch(db: D1Database, ids: string[]) {
  if (ids.length === 0) return { results: [] as TargetRow[] };
  const placeholders = ids.map(() => '?').join(',');
  return db.prepare(`SELECT * FROM targets WHERE id IN (${placeholders})`).bind(...ids).all<TargetRow>();
}

export async function advanceTarget(db: D1Database, id: string, nextRunAt: number) {
  await db.prepare(`UPDATE targets SET next_run_at = ? WHERE id = ?`).bind(nextRunAt, id).run();
}

export async function insertChecks(db: D1Database, checks: CheckRow[]) {
  if (checks.length === 0) return;
  const stmt = db.prepare(
    `INSERT INTO checks_raw (target_id, at, ok, latency_ms, error_kind, detail) VALUES (?, ?, ?, ?, ?, ?)`
  );
  await db.batch(checks.map((c) => stmt.bind(c.target_id, c.at, c.ok, c.latency_ms, c.error_kind, c.detail)));
}

export async function upsertRollup(db: D1Database, targetId: string, bucketMin: number, ok: boolean, latencyMs: number | null) {
  await db.prepare(
    `INSERT INTO checks_rollup (target_id, bucket_min, ok_count, fail_count, sum_ms, max_ms)
     VALUES (?, ?, ?, ?, ?, ?)
     ON CONFLICT(target_id, bucket_min) DO UPDATE SET
       ok_count = ok_count + excluded.ok_count,
       fail_count = fail_count + excluded.fail_count,
       sum_ms = sum_ms + excluded.sum_ms,
       max_ms = MAX(max_ms, excluded.max_ms)`
  ).bind(targetId, bucketMin, ok ? 1 : 0, ok ? 0 : 1, latencyMs ?? 0, latencyMs ?? 0).run();
}

export async function getOpenIncident(db: D1Database, targetId: string) {
  return db.prepare(`SELECT * FROM incidents WHERE target_id = ? AND closed_at IS NULL ORDER BY opened_at DESC LIMIT 1`).bind(targetId).first<IncidentRow>();
}

export async function insertIncident(db: D1Database, i: { id: string; targetId: string; openedAt: number; state: string; cause: string | null }) {
  await db.prepare(`INSERT INTO incidents (id, target_id, opened_at, closed_at, state, cause) VALUES (?, ?, ?, NULL, ?, ?)`)
    .bind(i.id, i.targetId, i.openedAt, i.state, i.cause).run();
}

export async function closeIncident(db: D1Database, id: string, at: number) {
  await db.prepare(`UPDATE incidents SET closed_at = ?, state = 'closed' WHERE id = ?`).bind(at, id).run();
}

export async function writeEvent(db: D1Database, e: { id: string; at: number; kind: string; actor: string | null; assetId: string | null; payload: string | null; ip: string | null }) {
  await db.prepare(`INSERT INTO events (id, at, kind, actor, asset_id, payload, ip) VALUES (?, ?, ?, ?, ?, ?, ?)`)
    .bind(e.id, e.at, e.kind, e.actor, e.assetId, e.payload, e.ip).run();
}

export async function getUserByEmail(db: D1Database, email: string) {
  return db.prepare(`SELECT * FROM users WHERE email = ?`).bind(email).first<UserRow>();
}
```

```ts
export type SourceKind = 'http' | 'tcp' | 'dns' | 'tls' | 'cloudflare' | 'agent';

export type ErrorKind = 'timeout' | 'dns' | 'tls' | 'status' | 'refused' | 'auth';

export interface Observation {
  targetId: string;
  at: number;
  ok: boolean;
```

```ts
export type Status = 'ok' | 'suspect' | 'down' | 'unknown';

export interface Machine {
  state: Status;
  failStreak: number;
  okStreak: number;
}

const FAIL_THRESHOLD = 3;
const OK_THRESHOLD = 2;

export function initial(): Machine {
  return { state: 'unknown', failStreak: 0, okStreak: 0 };
}

export function step(m: Machine, ok: boolean): Machine {
  if (ok) {
    const okStreak = Math.min(m.okStreak + 1, OK_THRESHOLD);
    if (okStreak >= OK_THRESHOLD) return { state: 'ok', failStreak: 0, okStreak };
    return { state: m.state === 'down' ? 'down' : 'suspect', failStreak: 0, okStreak };
  }
  const failStreak = Math.min(m.failStreak + 1, FAIL_THRESHOLD);
  // down 是粘性状态：已确认的故障只能被连续 2 次成功解除，
  // 单次失败不得把它打回 suspect（否则会重复产生 incident.opened）。
  if (m.state === 'down') return { state: 'down', failStreak, okStreak: 0 };
  return { state: failStreak >= FAIL_THRESHOLD ? 'down' : 'suspect', failStreak, okStreak: 0 };
}

export function toStatus(m: Machine): Status {
  return m.state;
}
```

```ts
import type { Observation, Target } from '../domain/observation';

export interface ProbeDeps {
  fetch: typeof fetch;
  now: () => number;
}

function classify(e: unknown): { errorKind: 'timeout' | 'dns' | 'tls' | 'refused'; message: string } {
  const err = e as { name?: string; message?: string };
  const msg = err?.message ?? String(e);
  if (err?.name === 'AbortError') return { errorKind: 'timeout', message: '请求超时' };
  if (/ENOTFOUND|EAI_AGAIN|getaddrinfo/i.test(msg)) return { errorKind: 'dns', message: msg };
  if (/certificate|SSL|TLS/i.test(msg)) return { errorKind: 'tls', message: msg };
  if (/ECONNREFUSED/i.test(msg)) return { errorKind: 'refused', message: msg };
  return { errorKind: 'timeout', message: msg };
}

export async function probeHttp(target: Target, deps: ProbeDeps): Promise<Observation> {
  const expectStatus = (target.config.expectStatus as number[] | undefined) ?? [200];
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), target.timeoutMs);
  const started = deps.now();
  try {
    const res = await deps.fetch(target.form, { method: 'GET', redirect: 'follow', signal: controller.signal });
    const latencyMs = deps.now() - started;
    const ok = expectStatus.includes(res.status);
    return {
      targetId: target.id,
      at: started,
      ok,
      latencyMs,
      ...(ok ? {} : { errorKind: 'status' as const, detail: { status: res.status } }),
    };
  } catch (e) {
    const { errorKind, message } = classify(e);
    return { targetId: target.id, at: started, ok: false, errorKind, detail: { message } };
  } finally {
    clearTimeout(timer);
  }
}
```

```ts
// src/scheduler/run.ts
import type { Env } from '../env';
import { listDueTargets, insertChecks, advanceTarget, upsertRollup, getOpenIncident, insertIncident, closeIncident, writeEvent } from '../db/queries';
import { resolveSource } from '../sources';
import { step, initial, type Machine } from '../domain/incident';
import type { Observation, Target } from '../domain/observation';

export const SHARD_LIMIT = 40;
export const CONCURRENCY = 6;

export interface TickDeps {
  now: () => number;
  fetch: typeof fetch;
}

async function mapLimit<T, R>(items: T[], limit: number, fn: (item: T) => Promise<R>): Promise<R[]> {
  const out: R[] = [];
  for (let i = 0; i < items.length; i += limit) {
    const slice = items.slice(i, i + limit);
    out.push(...(await Promise.all(slice.map(fn))));
  }
  return out;
}

function toTarget(t: { id: string; source: string; form: string; config: string; timeout_ms: number }): Target {
  return {
    id: t.id,
    source: t.source as Target['source'],
    form: t.form,
    config: JSON.parse(t.config) as Record<string, unknown>,
    timeoutMs: t.timeout_ms,
  };
}

export async function runTick(env: Env, deps: TickDeps): Promise<{ probed: number; skipped: number }> {
  const now = deps.now();
  const due = await listDueTargets(env.DB, now, SHARD_LIMIT);
  if (due.results.length === 0) return { probed: 0, skipped: 0 };

  const observations: Observation[] = await mapLimit(due.results, CONCURRENCY, async (row) => {
    try {
      return await resolveSource(toTarget(row), deps);
    } catch (e) {
      return { targetId: row.id, at: now, ok: false, errorKind: 'timeout' as const, detail: { message: String(e) } };
    }
  });

  await insertChecks(
    env.DB,
    observations.map((o) => ({
      target_id: o.targetId,
      at: o.at,
      ok: o.ok ? 1 : 0,
      latency_ms: o.latencyMs ?? null,
      error_kind: o.errorKind ?? null,
      detail: o.detail === undefined ? null : JSON.stringify(o.detail).slice(0, 1000),
    })),
  );

  for (const o of observations) {
    const bucketMin = Math.floor(o.at / 60_000);
    await upsertRollup(env.DB, o.targetId, bucketMin, o.ok, o.latencyMs ?? null);
  }

  for (const row of due.results) {
    await advanceTarget(env.DB, row.id, now + row.interval_sec * 1000);
  }

  for (const o of observations) {
    const open = await getOpenIncident(env.DB, o.targetId);
    const machine: Machine = open ? { state: open.state as Machine['state'], failStreak: 0, okStreak: 0 } : initial();
    const next = step(machine, o.ok);
    const assetId = due.results.find((r) => r.id === o.targetId)?.asset_id ?? null;

    if (next.state === 'down' && !open) {
      const id = crypto.randomUUID();
      await insertIncident(env.DB, { id, targetId: o.targetId, openedAt: o.at, state: 'down', cause: o.errorKind ?? null });
      await writeEvent(env.DB, {
        id: crypto.randomUUID(), at: o.at, kind: 'incident.opened', actor: null,
        assetId, payload: JSON.stringify({ incidentId: id, errorKind: o.errorKind ?? null }), ip: null,
      });
    }
    if (next.state === 'ok' && open) {
      await closeIncident(env.DB, open.id, o.at);
      await writeEvent(env.DB, {
        id: crypto.randomUUID(), at: o.at, kind: 'incident.closed', actor: null,
        assetId, payload: JSON.stringify({ incidentId: open.id, durationMs: o.at - open.opened_at }), ip: null,
      });
    }
  }

  return { probed: observations.length, skipped: 0 };
}
```

```ts
import type { Env } from '../env';
import { verifyAccessJwt } from './access';
import { getUserByEmail } from '../db/queries';

export interface Identity {
  email: string;
  role: 'owner' | 'ops' | 'viewer';
}

const CACHE_TTL_SEC = 60;

export async function getIdentity(req: Request, env: Env): Promise<Identity | null> {
  if (env.DEV_BYPASS_EMAIL) return { email: env.DEV_BYPASS_EMAIL, role: 'owner' };

  const token = req.headers.get('Cf-Access-Jwt-Assertion');
  if (!token) return null;

  const verified = await verifyAccessJwt(token, env);
  if (!verified) return null;

  const cacheKey = `identity:${verified.email}`;
  const cached = await env.KV.get(cacheKey);
  if (cached) return JSON.parse(cached) as Identity;

  const row = await getUserByEmail(env.DB, verified.email);
  if (!row) return null;

  const identity: Identity = { email: row.email, role: row.role as Identity['role'] };
  await env.KV.put(cacheKey, JSON.stringify(identity), { expirationTtl: CACHE_TTL_SEC });
  return identity;
}
```

```ts
import type { Identity } from '../auth/identity';

export type Danger = 'none' | 'confirm' | 'typed';

export interface ActionDef {
  id: string;
  label: string;
  roles: Identity['role'][];
  phase: 1 | 3;
  danger: Danger;
  reversible: boolean;
}

export const ACTIONS: ActionDef[] = [
  { id: 'probe.run', label: '立即探测', roles: ['owner', 'ops'], phase: 1, danger: 'none', reversible: true },
  { id: 'service.restart', label: '重启服务', roles: ['owner', 'ops'], phase: 3, danger: 'confirm', reversible: true },
  { id: 'service.logs', label: '查看日志', roles: ['owner', 'ops', 'viewer'], phase: 3, danger: 'none', reversible: true },
  { id: 'agent.restart', label: '重启 agent', roles: ['owner'], phase: 3, danger: 'typed', reversible: false },
];

export function actionsFor(role: Identity['role']): ActionDef[] {
  if (role === 'viewer') {
    return ACTIONS.filter((a) => a.danger === 'none' && a.roles.includes('viewer'));
  }
  return ACTIONS.filter((a) => a.roles.includes(role));
}

export class CommandNotAvailableError extends Error {
  constructor() {
    super('command channel is not available until phase 3');
    this.name = 'CommandNotAvailableError';
  }
}
```

```ts
  const res = await fetch(path, { headers: { accept: 'application/json' } });
  if (!res.ok) throw new Error(`GET ${path} → ${res.status}`);

}

export async function post<T>(path: string, body?: unknown): Promise<T> {
  const res = await fetch(path, {

    headers: { 'content-type': 'application/json' },
    body: body === undefined ? undefined : JSON.stringify(body),
  });
```

```tsx
export function Sparkline({ points, width = 60, height = 14 }: { points: number[]; width?: number; height?: number }) {
  if (points.length === 0) return <span style={{ color: 'var(--muted)' }}>·········</span>;
  const max = Math.max(...points, 1);
  const step = width / Math.max(points.length - 1, 1);
  const d = points.map((p, i) => `${i * step},${height - (p / max) * height}`).join(' ');
  return (
    <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`} preserveAspectRatio="none">
      <polyline points={d} fill="none" stroke="var(--accent)" strokeWidth="1" />
    </svg>
  );
}
```

```ts
// src/api/overview.ts
import { Hono } from 'hono';
import type { Env } from '../env';
import type { Vars } from '../auth/rbac';
import { requireRole } from '../auth/rbac';
import { listDueTargets } from '../db/queries';

export interface Overview {
  okCount: number; suspectCount: number; downCount: number; unknownCount: number; writeEstimate: number;
}

export async function buildOverview(env: Env): Promise<Overview> {
  const rows = await env.DB.prepare(
    `SELECT
       SUM(CASE WHEN i.id IS NOT NULL THEN 0 ELSE 0 END) AS dummy,
       COUNT(*) AS total
     FROM targets t LEFT JOIN incidents i ON i.target_id = t.id AND i.closed_at IS NULL`
  ).first<{ dummy: number | null; total: number }>();

  const down = await env.DB.prepare(`SELECT COUNT(*) AS n FROM incidents WHERE closed_at IS NULL`).first<{ n: number }>();
  const total = rows?.total ?? 0;
  const downCount = down?.n ?? 0;
  const writeEstimate = await estimateWrites(env);

  return {
    okCount: Math.max(total - downCount, 0),
    suspectCount: 0,
    downCount,
    unknownCount: 0,
    writeEstimate,
  };
}

async function estimateWrites(env: Env): Promise<number> {
  const row = await env.DB.prepare(
    `SELECT COUNT(*) AS n, COALESCE(SUM(86400 / NULLIF(interval_sec, 0)), 0) AS w FROM targets WHERE enabled = 1`
  ).first<{ n: number; w: number }>();
  return Math.round(row?.w ?? 0);
}

export const overviewApi = new Hono<{ Bindings: Env; Variables: Vars }>();

overviewApi.get('/', requireRole('owner', 'ops', 'viewer'), async (c) => c.json(await buildOverview(c.env)));
```

```tsx
        {allOk ? '全部正常。' : `${o.downCount} 个故障，${o.suspectCount} 个疑似。`}
      </h1>
      <div style={{ color: 'var(--muted)', marginBottom: 14 }}>
        {o.okCount} 正常 · {o.downCount} 故障 · {o.suspectCount} 疑似 · 每日写入估算 {o.writeEstimate.toLocaleString()} 行
      </div>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(4, 1fr)', gap: 8 }}>
        {[
          ['正常', o.okCount, 'var(--ok)'],
          ['疑似', o.suspectCount, 'var(--warn)'],
          ['故障', o.downCount, 'var(--danger)'],
          ['无数据', o.unknownCount, 'var(--muted)'],
        ].map(([label, value, color]) => (
          <div key={String(label)} className="card" style={{ padding: '8px 10px' }}>
            <div style={{ color: 'var(--muted)', fontSize: 10 }}>{label}</div>
            <div className="num" style={{ fontSize: 20, color: String(color) }}>{value}</div>
          </div>
        ))}
      </div>
    </div>
  );
}
```
